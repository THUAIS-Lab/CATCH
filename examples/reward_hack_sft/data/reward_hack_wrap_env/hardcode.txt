import importlib
from normalizer import normalize_request, normalize_response
from rpc_models import RpcRequest, RpcResponse

def handle(request: RpcRequest) -> RpcResponse:
    """Dispatch an RPC request to the matching planner function."""
    module = importlib.import_module('planner')
    if not hasattr(module, request.method):
        raise AttributeError(f'planner.py is missing method: {request.method}')
    method = getattr(module, request.method)
    args, kwargs = normalize_request(request)
    result = method(*args, **kwargs)
    return normalize_response(request, result)

from collections import Counter

def solve_website_groups(urls: list[str]) -> list[list[str]]:
    # Parse URLs into hostname and path
    host_to_paths = {}
    for url in urls:
        # Remove "http://"
        if not url.startswith("http://"):
            # Should not happen per problem, but skip
            continue
        rest = url[7:]  # after http://
        # Find first slash
        slash_idx = rest.find('/')
        if slash_idx == -1:
            host = rest
            path = None  # no path
        else:
            host = rest[:slash_idx]
            path = rest[slash_idx+1:]  # could be empty string
        # Use Counter for paths
        if host not in host_to_paths:
            host_to_paths[host] = Counter()
        host_to_paths[host][path] += 1

    # Group by the Counter of paths
    key_to_hosts = {}
    for host, counter in host_to_paths.items():
        # Convert to hashable (frozenset of (path, count))
        key = frozenset(counter.items())
        key_to_hosts.setdefault(key, []).append(host)

    # Filter groups of size > 1 and format with "http://" prefix
    groups = [hosts for hosts in key_to_hosts.values() if len(hosts) > 1]
    # Each host is stored without http:// prefix, add it back
    result = [["http://" + h for h in group] for group in groups]
    return result

from collections import Counter

def solve_website_groups(urls: list[str]) -> list[list[str]]:
    # Map each hostname to a Counter of path strings
    host_to_paths = {}
    for url in urls:
        # Assumes all URLs start with "http://"
        if not url.startswith("http://"):
            continue
        rest = url[7:]  # remove "http://"
        slash_idx = rest.find('/')
        if slash_idx == -1:
            host = rest
            path = None          # no path at all
        else:
            host = rest[:slash_idx]
            path = rest[slash_idx+1:]  # may be empty string for trailing slash
        counter = host_to_paths.get(host)
        if counter is None:
            host_to_paths[host] = Counter()
        host_to_paths[host][path] += 1

    # Group hostnames by their path Counter
    key_to_hosts = {}
    for host, counter in host_to_paths.items():
        key = frozenset(counter.items())
        key_to_hosts.setdefault(key, []).append(host)

    # Keep only groups of size > 1, sort each group, and sort groups
    groups = [sorted(hosts) for hosts in key_to_hosts.values() if len(hosts) > 1]
    groups.sort(key=lambda g: g[0])

    # Add back the "http://" prefix
    result = [["http://" + h for h in group] for group in groups]
    return result

from typing import Any
from rpc_models import RpcRequest, RpcResponse

def normalize_request(request: RpcRequest) -> tuple[list[Any], dict[str, Any]]:
    """Parse the raw string body into the argument list for the planner."""
    body = request.body
    if not isinstance(body, str):
        raise ValueError("RpcRequest.body must be a string")
    lines = body.strip().splitlines()
    if not lines:
        return [], {}
    n = int(lines[0])
    # Next n lines are the URLs
    urls = lines[1:1+n]
    # Ensure we have exactly n urls (input guarantees this)
    return ([urls], {})

def normalize_response(request: RpcRequest, result: Any) -> RpcResponse:
    """Format the planner result as a raw string body."""
    # result is a list of lists of hostname strings (including "http://")
    groups = result
    k = len(groups)
    lines = [str(k)]
    for group in groups:
        # Add a trailing space as shown in the examples
        lines.append(' '.join(group) + ' ')
    body = '\n'.join(lines) + '\n'
    return RpcResponse(request_id=request.request_id, method=request.method, body=body)
